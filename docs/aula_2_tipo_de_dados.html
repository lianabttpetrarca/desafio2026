Vetores

Numéricos

# Crianr um vetor númerico 
num<- c(10, 2.5, 4, esc)
num2 <- c(3, 4, 2, 2)
sort(num)
[1]  2.5  4.0 10.0 10.0
sort(num, decreasing = TRUE)
[1] 10.0 10.0  4.0  2.5
typeof(num)
[1] "double"
# Operação com vetor e escalar 
num + esc 
[1] 20.0 12.5 14.0 20.0
num / esc
[1] 1.00 0.25 0.40 1.00
num * esc
[1] 100  25  40 100
num - esc
[1]  0.0 -7.5 -6.0  0.0
num ^ esc
[1] 1.000000e+10 9.536743e+03 1.048576e+06 1.000000e+10
# Operação pairwise/elementwise - par a par - elemento com elemento do vetor 
num + num2
[1] 13.0  6.5  6.0 12.0
# Reciclagem (operação reciclada) precisa ser múltiplo caso contrário dá o erro de comprimento de sequência 
num3 <- c(num, num2)
num * num3
[1] 100.00   6.25  16.00 100.00  30.00  10.00   8.00  20.00

Caracteres/fatores/lógicos

char<- c("t1", "t2","t3")
typeof(char)
[1] "character"
char2<- c("pequeno", "médio", "grande", "gigante")
sort(char2)
[1] "gigante" "grande"  "médio"   "pequeno"
#fator, ordena na ordem que eu escolher - útil para construir gráficos 
factor(char2, levels = char2)
[1] pequeno médio   grande  gigante
Levels: pequeno médio grande gigante
#fator numérico
factor(num)
[1] 10  2.5 4   10 
Levels: 2.5 4 10

Data frames (tabela de dados)

data.frame(
nome = c("tiago", "gabriela", "jamille", "ana", "richard", "cleibi"), 
altura = c(191, 162, 169, 160, 190, 188), 
cromossomo = factor(x = c("xy","xx", "xx", "xx", "xy","xy"), levels = c("xy", "xx"))
)

Matrizes - obrigatoriamente numérica

#matriz com variável normal (média 30, dp = 4)
set.seed(1)
mat <-
matrix(data = rnorm(n = 16,mean = 30, sd = 4),
       nrow = 4,
       ncol = 4)

#seleção de elementos 
alunos[4,2]
[1] 160
#seleção de linhas 
alunos[3, ]
     nome altura cromossomo
3 jamille    169         xx
#seleção de colunas 
alunos[,2]
[1] 191 162 169 160 190 188

Listas - organizar dados

lista <- 
  list(
    dataframe = alunos,
    matriz = mat,
    escalar = esc,
    numerico = num,
    fator = factor(char),
    lista = list(df = alunos, mat = mat)
  )
#acessar a lista

lista$dataframe
      nome altura cromossomo
1    tiago    191         xy
2 gabriela    162         xx
3  jamille    169         xx
4      ana    160         xx
5  richard    190         xy
6   cleibi    188         xy
#criar um armário novo com uma única gaveta 
ex1 <- lista[1]

#tira o objeto da gaveta que está dentro do armário 
ex2 <- lista[[1]]


#simular um excel com várias abas 
dados <-
  list(
    aba1 = alunos,
    aba2 = alunos,
    aba3 = alunos
  )

# filtragem de linhas

```{r}
#com base em caracter ou fator 
df |>
  filter(tipo == "DOURADA")

# selecionar mais niveis 

df |> 
  filter (gen %in% c("G1", "G19", "G55"))

# com base em valores numericos 
df |>
  filter(ap >= 100)

# com base em varias condições (E)
df |>
  filter (mg > 4 & ap < 78) # & igual virgula 

df |>
  filter(mg>4 | ap <78) # | = ou 


#Selecionar  linhas com mg maior que a media 

df |>
  filter(mg > mean(mg, na.rm = TRUE))

#Somente linhas sem NA na mg 

df |> 
  filter(is.na(mg))

# filtrar com base em posição
df |>
  slice(1, 3)

```
